(function() {
    const now = moment(); 

    let greeting;
    const hour = now.hour();

    if (hour < 11) {
        greeting = "С добрым утром! Впереди продуктивный день";
    } else if (hour < 18) {
        greeting = "Полдня уже прошло — поднажмём!";
    } else if (hour < 22) {
        greeting = "Вечер - потихоньку подводим итоги!";
    } else {
        greeting = "Заканчиваем работу и отправляем отчёт, товарищ!";
    }

    const currentTime = now.format("DD-MMMM-YYYY | HH:mm | Бета Омега Альфа");

    dv.header(1, `${greeting}`);
    dv.paragraph(`<span style="font-size: 1.4em; color: #666;">${currentTime}</span>`);

})();

>[!general aims]- Основные положения хранилища
>>[!point]- Принцип конфликтности
>>Ни в коем случае нельзя начинать войну, когда это нецелесообразно или бессмысленно. Нет смысла тратить ценнейшее время и ресурсы на пустяки: на твоих руках итак хорошие карты. Кроме того, преступно забывать изначальные заповеди
>
>>[!point]- Принцип защиты
>>Если враг совершает свою нападение, то он должен получить такой отпор, чтобы у него и мысли не было появиться перед тобой вновь. В этой борьбе важно выйти на своих условиях
>
>>[!point]- Принцип напора
>>Вера ведет меня вперед. Каждый раз, когда я хочу сдаться, необходимо напоминать себе, за что я сражаюсь. Тот образ, отражение
>
>>[!point]- Принцип участия
>>Необходимо принимать участие в жизни друзей. Необходимо слушать и молчать, когда это уместно. И, конечно, недопустимо отравлять жизнь других людей, предавать их даже в мелочах и перестать раздрожать  
>
>>[!point]- Принцип улучшения
>>Стремление к самосовершенствованию как в физическом, так и умственном плане. Обеспечение выполнения актуальных задач и правильное ведение дел 
>
>>[!point]- Принцип полезности
>>Некоторые вещи просто полезны. Подлости могут быть оправданы
