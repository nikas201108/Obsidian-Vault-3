Алгоритм для одной кучи:
```python
def P(n, score):
    if score < 40 and n == 3 and any([score + 1 >= 40, score + 4 >= 40, score*2 >= 40]):
        return True
    if n == 3:
        return False
    if n == 1 and any([score + 1 >= 40, score + 4 >= 40, score*2 >= 40]):
        return False
    return any([I(n + 1, score + 1), I(n + 1, score + 4), I(n + 1, score * 2)])
    
def I(n, score):
    return all([P(n+1,  score + 1), P(n+1, score + 4), P(n+1, score*2)])

for x in range(1, 40):
    if P(1, x):
        print(x)
```
